}

type review2GateWorker struct { calls int }
func(w *review2GateWorker) Execute(context.Context,goaldrive.WorkerRequest)(goaldrive.WorkerResult,error){w.calls++;return goaldrive.WorkerResult{Outcome:goaldrive.OutcomeNoProgress,EndHead:"head"},nil}
func TestReview2ExplicitGateObjectiveReachesWorker(t *testing.T){
 h:=newGateHarness(t);w:=&review2GateWorker{}
 c:=goaldrive.Controller{Ledger:h.ledger,Worker:w,SafetyActivation:lifecycleSafetyActivation{getenv:h.env}}
 _,err:=c.ExecuteTurn(h.ctx,goaldrive.TurnRequest{GoalID:repairGoalID,GoalVersion:"2",InvocationID:"explicit-gate",TurnID:"explicit-gate",GraphID:"graph",GraphVersion:"1",StartHead:"head",Mode:goaldrive.ModeSupervised,GoalBaseline:&h.successor,ChildObjective:"gate",Repository:contracts.RepositorySynced})
 if w.calls!=1{t.Fatalf("worker calls=%d err=%v",w.calls,err)}
 t.Logf("explicit gate objective with candidates loaded later reaches worker: calls=%d err=%v",w.calls,err)
}
