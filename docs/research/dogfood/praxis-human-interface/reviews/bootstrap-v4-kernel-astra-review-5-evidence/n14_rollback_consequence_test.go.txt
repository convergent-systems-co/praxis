// PASS means the counterexample reproduced. This extends Repair #4's preserved
// A2 helper probe through restart and the real gate-completion consequence.
func TestReview5N14ReplayedLivenessCreatesNewGovernedCompletion(t *testing.T) {
	h := newGateHarness(t)
	repo, db, err := openGovernedRepository(h.ctx, h.env)
	if err != nil {
		t.Fatal(err)
	}
	raw, err := h.ledger.LoadCompletions(h.ctx, repairGoalID, "2")
	if err != nil || len(raw) != 1 {
		t.Fatalf("fixture producer: %v %+v", err, raw)
	}
	ledger := goaldrive.Ledger{Store: state.NewSQLiteEventStore(db), Actor: contracts.PrincipalRef{ID: "controller", Kind: "controller"}}
	if err := ledger.RecordCompletion(h.ctx, raw[0]); err != nil {
		t.Fatal(err)
	}
	request := h.pendingGate(ledger, "review5-pending")
	requestDigest, _ := request.Digest()
	if _, err := h.decide(requestDigest, "approve", "finish and reconcile", "DECIDE-APPROVE "+requestDigest+" ALTERNATIVE finish and reconcile"); err != nil {
		t.Fatal(err)
	}

	// Copy only the already-sealed liveness row while it is current. The
	// adversarial phase below never invokes Crypto or uses KeyRef.
	var ns, id, version, digest, sensitivity, profile, created string
	var envelope []byte
	var expires *string
	if err := db.QueryRowContext(h.ctx, `SELECT namespace,object_id,object_version,object_digest,sensitivity,crypto_profile,envelope_json,created_at,expires_at FROM secure_blobs WHERE namespace='authority_decision_live' AND object_id=? AND object_version=?`, request.ID, request.Version).Scan(&ns, &id, &version, &digest, &sensitivity, &profile, &envelope, &created, &expires); err != nil {
		t.Fatal(err)
	}
	decision, err := repo.LoadAuthorityDecision(h.ctx, request.ID, request.Version, time.Now().UTC())
	if err != nil {
		t.Fatal(err)
	}
	decisionDigest, _ := decision.Digest()
	now := time.Now().UTC()
	revocation := contracts.AuthorityRevocation{RequestID: request.ID, RequestVersion: request.Version, DecisionRef: decision.DecisionRef, DecisionVersion: decision.DecisionVersion, DecisionDigest: decisionDigest, RevocationRef: "review5-revoke", RevocationVersion: "1", RevokedBy: h.root.Principal, AuthorityDigest: h.root.AuthorityModelDigest, EffectiveAt: now, Reason: "owner withdrew gate authority"}
	if err := repo.SaveAuthorityRevocation(h.ctx, request.ID, request.Version, revocation, now, nil); err != nil {
		t.Fatal(err)
	}
	if _, err := repo.LoadAuthorityDecision(h.ctx, request.ID, request.Version, time.Now().UTC()); !errors.Is(err, goalstore.ErrAuthorityDecisionRevoked) {
		t.Fatalf("control: revocation must refuse: %v", err)
	}
	if _, err := h.drive(ledger, "review5-revoked"); !errors.Is(err, goalstore.ErrAuthorityDecisionRevoked) {
		t.Fatalf("control: revoked drive must refuse: %v", err)
	}
	before, err := ledger.LoadCompletions(h.ctx, repairGoalID, "2")
	if err != nil || len(before) != 1 {
		t.Fatalf("control: revoked drive changed consequence: %v %+v", err, before)
	}

	// Roll the authority view back: remove the later revocation and replay the
	// byte-exact earlier liveness row.
	if _, err := db.ExecContext(h.ctx, `DELETE FROM secure_blobs WHERE namespace='authority_revocation' AND object_id=? AND object_version=?`, request.ID, request.Version); err != nil {
		t.Fatal(err)
	}
	if _, err := db.ExecContext(h.ctx, `INSERT INTO secure_blobs(namespace,object_id,object_version,object_digest,sensitivity,crypto_profile,envelope_json,created_at,expires_at) VALUES(?,?,?,?,?,?,?,?,?)`, ns, id, version, digest, sensitivity, profile, envelope, created, expires); err != nil {
		t.Fatal(err)
	}
	if err := db.Close(); err != nil {
		t.Fatal(err)
	}

	// Restart: fresh database, repository, ledger and controller views.
	repo, db, err = openGovernedRepository(h.ctx, h.env)
	if err != nil {
		t.Fatal(err)
	}
	defer db.Close()
	baseline, err := repo.Load(h.ctx, repairGoalID, "2", time.Now().UTC())
	if err != nil {
		t.Fatal(err)
	}
	h.successor = baseline
	ledger = goaldrive.Ledger{Store: state.NewSQLiteEventStore(db), Actor: contracts.PrincipalRef{ID: "controller", Kind: "controller"}}
	if _, err := repo.LoadAuthorityDecision(h.ctx, request.ID, request.Version, time.Now().UTC()); err != nil {
		t.Fatalf("rollback did not restore the decision: %v", err)
	}
	if _, err := h.drive(ledger, "review5-after-replay"); !errors.Is(err, goaldrive.ErrHumanAuthorityGate) {
		t.Fatalf("expected the restored decision to create the gate completion: %v", err)
	}
	after, err := ledger.LoadCompletions(h.ctx, repairGoalID, "2")
	if err != nil || len(after) != 2 {
		t.Fatalf("new durable completion missing: %v %+v", err, after)
	}
	effective, err := goaldrive.LoadEffectiveCompletions(h.ctx, ledger, repo, &baseline, repairGoalID, "2")
	if err != nil || len(effective.Effective) != 2 || len(effective.Historical) != 0 {
		t.Fatalf("rollback completion not effective: %v %+v", err, effective)
	}
	assessment, err := goaldrive.AssessGoalCompletion(baseline, effective.Effective)
	if err != nil || !assessment.AllUnitsComplete {
		t.Fatalf("rollback did not establish eligibility: %v %+v", err, assessment)
	}
	t.Logf("COUNTEREXAMPLE: copied sealed liveness replay restored revoked authority after restart; controller minted a NEW sealed completion, completions %d -> %d, AllUnitsComplete=%v", len(before), len(after), assessment.AllUnitsComplete)
}

// A3 is reachable by the same delete-only SQLite writer used in Review #4.
// Erasing the classification plus the two artifact families consulted by the
// fallback leaves other authenticated acceptance state intact but makes the
// Goal appear legacy. The supported import then admits stripped work.
func TestReview5N15ErasingClassificationSourcesPermitsDowngradedExecution(t *testing.T) {
	h := newGateHarness(t)
	repo, db, err := openGovernedRepository(h.ctx, h.env)
	if err != nil {
		t.Fatal(err)
	}
	downgraded := strippedGeneration(h, "3")
	// Keep the surviving plan-less generation 1 as the import predecessor.
	downgraded.PredecessorDigest = h.baseline.Digest
	downgraded.Digest = ""
	path := writeBaselineImportDocument(t, h.dir, "review5-total-erasure.json", downgraded, "")
	h.removeActivation()
	if _, err := h.run("goals-lifecycle", "--operation=import", "--input="+path); err == nil {
		t.Fatal("control: stripped import succeeded while safety evidence survived")
	}

	// Delete only the positive facts GoalSafetyKernel consults: its direct
	// classification, its safety proposal, and its attached safety generation.
	// Other authenticated records, including the acceptance, remain present.
	deletions := []struct {
		query string
		args  []any
	}{
		{`DELETE FROM secure_blobs WHERE namespace='goal_safety_classification' AND object_id=?`, []any{repairGoalID}},
		{`DELETE FROM secure_blobs WHERE namespace='work_plan_proposal'`, nil},
		{`DELETE FROM secure_blobs WHERE namespace='goal_baseline' AND object_id=? AND object_version='2'`, []any{repairGoalID}},
	}
	for _, deletion := range deletions {
		result, err := db.ExecContext(h.ctx, deletion.query, deletion.args...)
		if err != nil {
			t.Fatal(err)
		}
		if rows, err := result.RowsAffected(); err != nil || rows < 1 {
			t.Fatalf("expected safety evidence deletion, rows=%d err=%v", rows, err)
		}
	}
	if err := db.Close(); err != nil {
		t.Fatal(err)
	}

	// Restart and use only supported production surfaces after the deletion.
	if _, err := h.run("goals-lifecycle", "--operation=import", "--input="+path); err != nil {
		t.Fatalf("stripped import after safety-history erasure: %v", err)
	}
	repo, db, err = openGovernedRepository(h.ctx, h.env)
	if err != nil {
		t.Fatal(err)
	}
	defer db.Close()
	classified, err := repo.GoalSafetyClassified(h.ctx, repairGoalID)
	if err != nil || classified {
		t.Fatalf("erased Goal did not become legacy: classified=%v err=%v", classified, err)
	}
	baseline, err := repo.Load(h.ctx, repairGoalID, "3", time.Now().UTC())
	if err != nil {
		t.Fatal(err)
	}
	ctrl := newGoalDriveController(h.ledger, nil, nil, repo, 0, h.env)
	worker := &recordingWorker{}
	ctrl.Worker = worker
	_, driveErr := ctrl.ExecuteTurn(h.ctx, goaldrive.TurnRequest{GoalID: repairGoalID, GoalVersion: "3", InvocationID: "review5-erasure", TurnID: "review5-erasure", GraphID: "g", GraphVersion: "1", StartHead: "head", ChildObjective: "one", Mode: goaldrive.ModeSupervised, GoalBaseline: &baseline, Repository: contracts.RepositorySynced})
	if worker.calls != 1 {
		t.Fatalf("downgraded worker not reached: calls=%d err=%v", worker.calls, driveErr)
	}
	t.Logf("COUNTEREXAMPLE: delete-only erasure of the classification sources made surviving authenticated acceptance state irrelevant; stripped import succeeded and worker ran with activation absent, calls=%d final error=%v", worker.calls, driveErr)
}
